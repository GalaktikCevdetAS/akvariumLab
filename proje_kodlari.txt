'use client';

import React, { useRef, useMemo } from 'react';
import { Canvas, useFrame } from '@react-three/fiber';
import { OrbitControls, Environment } from '@react-three/drei';
import SceneObjectRegistry from './SceneObjectRegistry';
import { useAquariumStore } from '../../store/useAquariumStore';
import { getTankDimensionsCm } from '../../lib/simulation/ecosystemEngine';
import * as THREE from 'three';

function AnimatedAquariumWater({
  tankWidth,
  tankHeight,
  tankDepth,
}: {
  tankWidth: number;
  tankHeight: number;
  tankDepth: number;
}) {
  const { waterVisuals } = useAquariumStore((state) => state.ecosystemStatus);

  const matRef = useRef<THREE.MeshStandardMaterial>(null);
  const targetColor = useMemo(() => new THREE.Color(waterVisuals.color), [waterVisuals.color]);

  useFrame(({ clock }, delta) => {
    if (!matRef.current) return;
    const t = clock.getElapsedTime();
    matRef.current.color.lerp(targetColor, delta * 2.2);

    const pulse = Math.sin(t * 1.5) * 0.012;
    const currentTargetOpacity = Math.max(0.08, waterVisuals.opacity + pulse);

    matRef.current.opacity = THREE.MathUtils.lerp(
      matRef.current.opacity,
      currentTargetOpacity,
      delta * 2.2
    );
    matRef.current.roughness = THREE.MathUtils.lerp(
      matRef.current.roughness,
      waterVisuals.roughness,
      delta * 2.2
    );
  });

  return (
    <mesh position={[0, -0.1, 0]}>
      <boxGeometry args={[tankWidth - 0.08, tankHeight - 0.25, tankDepth - 0.08]} />
      <meshStandardMaterial
        ref={matRef}
        color={waterVisuals.color}
        transparent
        opacity={waterVisuals.opacity}
        roughness={waterVisuals.roughness}
        depthWrite={false}
      />
    </mesh>
  );
}

function FpsTracker({ onFpsUpdate }: { onFpsUpdate: (fps: number) => void }) {
  const frameCount = useRef(0);
  const elapsed = useRef(0);

  useFrame((_, delta) => {
    frameCount.current += 1;
    elapsed.current += delta;

    if (elapsed.current >= 0.5) {
      const avgFps = Math.round(frameCount.current / elapsed.current);
      onFpsUpdate(avgFps);
      frameCount.current = 0;
      elapsed.current = 0;
    }
  });

  return null;
}

interface AquariumSceneProps {
  onFpsUpdate?: (fps: number) => void;
}

export default function AquariumScene({ onFpsUpdate }: AquariumSceneProps) {
  const tankCapacity = useAquariumStore((state) => state.tankCapacity);
  const dims = useMemo(() => getTankDimensionsCm(tankCapacity), [tankCapacity]);

  return (
    <div className="w-full h-full bg-[#060d18]" style={{ width: '100%', height: '100%', position: 'relative' }}>
      <Canvas
        camera={{ position: [0, 1.35, 5.2], fov: 42 }}
        gl={{ antialias: true, alpha: false }}
      >
        <color attach="background" args={['#060d18']} />

        {onFpsUpdate && <FpsTracker onFpsUpdate={onFpsUpdate} />}

        <ambientLight intensity={0.75} />
        <directionalLight position={[5, 10, 5]} intensity={1.3} castShadow />
        <pointLight position={[-4, 4, -4]} intensity={0.6} color="#38bdf8" />

        {/* 3D Cam Akvaryum TankÄ± GÃ¶vdesi (50L: 50x35x30 cm Ã¶lÃ§ekli) */}
        <mesh position={[0, 0, 0]}>
          <boxGeometry args={[dims.sceneWidth, dims.sceneHeight, dims.sceneDepth]} />
          <meshPhysicalMaterial
            color="#38bdf8"
            transmission={0.94}
            opacity={1}
            transparent
            roughness={0.08}
            ior={1.5}
            side={THREE.BackSide}
          />
        </mesh>

        {/* Dinamik Su Hacmi */}
        <AnimatedAquariumWater
          tankWidth={dims.sceneWidth}
          tankHeight={dims.sceneHeight}
          tankDepth={dims.sceneDepth}
        />

        {/* Sahne Objeleri */}
        <SceneObjectRegistry
          tankWidth={dims.sceneWidth}
          tankHeight={dims.sceneHeight}
          tankDepth={dims.sceneDepth}
        />

        <OrbitControls
          target={[0, 0.1, 0]}
          enablePan={true}
          enableZoom={true}
          minDistance={2.5}
          maxDistance={12}
          maxPolarAngle={Math.PI / 2 + 0.1}
        />

        <Environment preset="city" />
      </Canvas>
    </div>
  );
}

function SnailGroup({
  specs,
  innerHalfW,
  innerHalfD,
  substrateTopY,
  waterTopY,
  isCritical,
}: {
  specs: SnailSpec[];
  innerHalfW: number;
  innerHalfD: number;
  substrateTopY: number;
  waterTopY: number;
  isCritical: boolean;
}) {
  const shellRef = useRef<THREE.InstancedMesh>(null);
  const bodyRef = useRef<THREE.InstancedMesh>(null);

  const elmaGeo = useMemo(() => buildSnailGeometries('elma'), []);
  const neriteGeo = useMemo(() => buildSnailGeometries('nerite'), []);

  const dummy = useMemo(() => new THREE.Object3D(), []);
  const tempColor = useMemo(() => new THREE.Color(), []);

  // Her salyangozun akvaryum Ä°Ã‡Ä° sÄ±nÄ±rlarÄ±nda kalacaÄŸÄ± gÃ¼venli baÅŸlangÄ±Ã§ noktasÄ±
  const agents = useMemo(() => {
    const safeMinY = substrateTopY + 0.015;
    const safeMaxY = Math.max(safeMinY + 0.02, waterTopY - 0.022);
    const margin = 0.024;

    return specs.map((spec, idx) => {
      // 0: Arka cam iÃ§ yÃ¼zeyi, 1: Sol cam iÃ§ yÃ¼zeyi, 2: SaÄŸ cam iÃ§ yÃ¼zeyi, 3: Taban kumu
      const surface = idx % 4;
      const randA = ((idx * 37) % 100) / 100;
      const randB = ((idx * 73) % 100) / 100;

      let x = 0;
      let y = safeMinY + randB * (safeMaxY - safeMinY);
      let z = 0;

      if (surface === 0) {
        // Arka cam iÃ§ yÃ¼zeyi (z = -innerHalfD + margin)
        x = (randA * 2 - 1) * Math.max(0.02, innerHalfW - margin);
        z = -innerHalfD + margin;
      } else if (surface === 1) {
        // Sol cam iÃ§ yÃ¼zeyi
        x = -innerHalfW + margin;
        z = (randA * 2 - 1) * Math.max(0.02, innerHalfD - margin);
      } else if (surface === 2) {
        // SaÄŸ cam iÃ§ yÃ¼zeyi
        x = innerHalfW - margin;
        z = (randA * 2 - 1) * Math.max(0.02, innerHalfD - margin);
      } else {
        // Taban kumu Ã¼stÃ¼
        x = (randA * 2 - 1) * Math.max(0.02, innerHalfW - margin);
        z = (randB * 2 - 1) * Math.max(0.02, innerHalfD - margin);
        y = substrateTopY + 0.004;
      }

      return {
        surface,
        x,
        y,
        z,
        vx: (randA - 0.5) * 0.006,
        vy: (randB - 0.5) * 0.005,
        vz: (randA - 0.5) * 0.006,
        phase: idx * 1.7,
      };
    });
  }, [specs, innerHalfW, innerHalfD, substrateTopY, waterTopY]);

  useLayoutEffect(() => {
    if (!shellRef.current || !bodyRef.current) return;
    specs.forEach((spec, i) => {
      tempColor.set(spec.color);
      shellRef.current!.setColorAt(i, tempColor);
      bodyRef.current!.setColorAt(i, spec.species === 'nerite' ? '#3f3f46' : '#eab308');
    });
    if (shellRef.current.instanceColor) shellRef.current.instanceColor.needsUpdate = true;
    if (bodyRef.current.instanceColor) bodyRef.current.instanceColor.needsUpdate = true;
  }, [specs, tempColor]);

  useFrame((_, delta) => {
    if (!shellRef.current || !bodyRef.current || specs.length === 0) return;

    const dt = Math.min(delta, 0.05);
    const safeMinY = substrateTopY + 0.012;
    const safeMaxY = Math.max(safeMinY + 0.02, waterTopY - 0.02);
    const margin = 0.022;
    const maxX = Math.max(0.02, innerHalfW - margin);
    const maxZ = Math.max(0.02, innerHalfD - margin);

    agents.forEach((a, i) => {
      const spec = specs[i];

      if (isCritical) {
        // Toksik suda tabana dÃ¼ÅŸÃ¼p yan yatma
        a.y = THREE.MathUtils.lerp(a.y, substrateTopY + 0.005, dt * 1.5);
        dummy.position.set(a.x, a.y, a.z);
        dummy.rotation.set(0, a.phase, Math.PI * 0.45);
      } else {
        if (a.surface === 0) {
          // Arka cam iÃ§ yÃ¼zeyi
          a.x += a.vx * dt;
          a.y += a.vy * dt;
          if (a.x > maxX || a.x < -maxX) a.vx *= -1;
          if (a.y > safeMaxY || a.y < safeMinY) a.vy *= -1;
          a.x = THREE.MathUtils.clamp(a.x, -maxX, maxX);
          a.y = THREE.MathUtils.clamp(a.y, safeMinY, safeMaxY);
          a.z = -innerHalfD + margin;

          dummy.position.set(a.x, a.y, a.z);
          dummy.rotation.set(Math.PI / 2, Math.atan2(a.vx, a.vy), 0);
        } else if (a.surface === 1) {
          // Sol cam iÃ§ yÃ¼zeyi
          a.z += a.vz * dt;
          a.y += a.vy * dt;
          if (a.z > maxZ || a.z < -maxZ) a.vz *= -1;
          if (a.y > safeMaxY || a.y < safeMinY) a.vy *= -1;
          a.z = THREE.MathUtils.clamp(a.z, -maxZ, maxZ);
          a.y = THREE.MathUtils.clamp(a.y, safeMinY, safeMaxY);
          a.x = -innerHalfW + margin;

          dummy.position.set(a.x, a.y, a.z);
          dummy.rotation.set(0, 0, -Math.PI / 2);
        } else if (a.surface === 2) {
          // SaÄŸ cam iÃ§ yÃ¼zeyi
          a.z += a.vz * dt;
          a.y += a.vy * dt;
          if (a.z > maxZ || a.z < -maxZ) a.vz *= -1;
          if (a.y > safeMaxY || a.y < safeMinY) a.vy *= -1;
          a.z = THREE.MathUtils.clamp(a.z, -maxZ, maxZ);
          a.y = THREE.MathUtils.clamp(a.y, safeMinY, safeMaxY);
          a.x = innerHalfW - margin;

          dummy.position.set(a.x, a.y, a.z);
          dummy.rotation.set(0, 0, Math.PI / 2);
        } else {
          // Taban kumu Ã¼zeri
          a.x += a.vx * dt;
          a.z += a.vz * dt;
          if (a.x > maxX || a.x < -maxX) a.vx *= -1;
          if (a.z > maxZ || a.z < -maxZ) a.vz *= -1;
          a.x = THREE.MathUtils.clamp(a.x, -maxX, maxX);
          a.z = THREE.MathUtils.clamp(a.z, -maxZ, maxZ);
          a.y = substrateTopY + 0.004;

          dummy.position.set(a.x, a.y, a.z);
          dummy.rotation.set(0, Math.atan2(a.vx, a.vz), 0);
        }
      }

      dummy.scale.setScalar(spec.scale);
      dummy.updateMatrix();
      shellRef.current!.setMatrixAt(i, dummy.matrix);
      bodyRef.current!.setMatrixAt(i, dummy.matrix);
    });

    shellRef.current.instanceMatrix.needsUpdate = true;
    bodyRef.current.instanceMatrix.needsUpdate = true;
  });

  if (specs.length === 0) return null;

  return (
    <group>
      {/* Elma + Nerite aynÄ± instanced mesh Ã¼zerinden vertexColors ile ayrÄ±ÅŸÄ±r */}
      <instancedMesh
        ref={shellRef}
        key={`snail-shell-${specs.length}`}
        args={[
          specs[0].species === 'nerite'
            ? neriteGeo.shellGeometry
            : elmaGeo.shellGeometry,
          undefined,
          specs.length,
        ]}
        castShadow
      >
        <meshStandardMaterial vertexColors roughness={0.35} metalness={0.08} />
      </instancedMesh>

      <instancedMesh
        ref={bodyRef}
        key={`snail-body-${specs.length}`}
        args={[elmaGeo.bodyGeometry, undefined, specs.length]}
      >
        <meshStandardMaterial roughness={0.75} />
      </instancedMesh>
    </group>
  );
}

export default function SceneObjectRegistry({
  tankItems,
  tankWidth,
  tankHeight,
  tankDepth,
  waterLevelRatio = 0.9,
  isCritical = false,
}: SceneObjectRegistryProps) {
  const glassThickness = 0.012;
  const innerW = tankWidth - glassThickness * 2;
  const innerD = tankDepth - glassThickness * 2;
  const innerHalfW = innerW / 2;
  const innerHalfD = innerD / 2;

  const bottomInnerY = -tankHeight / 2 + glassThickness;
  const waterTopY = -tankHeight / 2 + tankHeight * waterLevelRatio;

  // 1. KUM HESABI
  const { totalSandKg, sandType, sandHeight } = useMemo(() => {
    let kg = 0;
    let type: 'silis' | 'hagen' | 'aktif-toprak' = 'silis';

    tankItems.forEach((item) => {
      if (item.product.category === 'taban' || item.product.category === 'kum') {
        const unitKg = item.product.weight_kg ?? 5;
        kg += unitKg * item.quantity;
        if (item.product.slug.includes('hagen')) type = 'hagen';
        else if (item.product.slug.includes('aktif')) type = 'aktif-toprak';
      }
    });

    const vol = kg * 0.00065;
    const h = Math.min(tankHeight * 0.35, vol / Math.max(0.01, innerW * innerD));
    return { totalSandKg: kg, sandType: type, sandHeight: h };
  }, [tankItems, innerW, innerD, tankHeight]);

  const substrateTopY = bottomInnerY + sandHeight;

  // 2. SALYANGOZLAR (ELMA SALYANGOZU + ZEBRA NERÄ°TE)
  const snailSpecs = useMemo(() => {
    const list: SnailSpec[] = [];

    tankItems.forEach((item) => {
      const slug = item.product.slug.toLowerCase();
      const name = item.product.name.toLowerCase();
      const isNerite = slug.includes('nerite') || name.includes('nerite');
      const isApple = slug.includes('elma') || name.includes('elma');
      const isGenericSnail =
        item.product.sub_category === 'salyangozlar' ||
        slug.includes('salyangoz') ||
        name.includes('salyangoz');

      if (!isNerite && !isApple && !isGenericSnail) return;

      for (let q = 0; q < item.quantity; q++) {
        if (isNerite) {
          list.push({
            species: 'nerite',
            scale: 0.95,
            color: '#ffffff', // vertexColors siyah-sarÄ± zebra desenini taÅŸÄ±r
          });
        } else {
          list.push({
            species: 'elma',
            scale: 1.25,
            color: slug.includes('mavi')
              ? '#60a5fa'
              : slug.includes('fildisi') || slug.includes('beyaz')
              ? '#fef3c7'
              : '#fbbf24',
          });
        }
      }
    });

    return list;
  }, [tankItems]);

  // 3. BÄ°TKÄ°LER (TÃœM EKLENEN BÄ°TKÄ° KALEMLERÄ°)
  const plantPositions = useMemo(() => {
    const positions: Array<{ x: number; z: number; scale: number; color: string }> = [];
    let seed = 1;

    tankItems.forEach((item) => {
      if (item.product.category !== 'bitkiler') return;
      const slug = item.product.slug.toLowerCase();

      for (let q = 0; q < item.quantity; q++) {
        seed += 19;
        const rx = ((seed * 31) % 100) / 100;
        const rz = ((seed * 53) % 100) / 100;
        positions.push({
          x: (rx * 2 - 1) * Math.max(0.04, innerHalfW - 0.06),
          z: (rz * 2 - 1) * Math.max(0.04, innerHalfD - 0.06),
          scale: 0.85 + (seed % 5) * 0.1,
          color: slug.includes('alternanthera') || slug.includes('kirmizi')
            ? '#b91c1c'
            : slug.includes('moss')
            ? '#15803d'
            : '#16a34a',
        });
      }
    });

    return positions;
  }, [tankItems, innerHalfW, innerHalfD]);

  // 4. FÄ°LTRE / MOTOR KONTROLÃœ (TÃœM MOTOR ALT KATEGORÄ°SÄ°)
  const hasFilterMotor = useMemo(() => {
    return tankItems.some(
      (i) =>
        i.quantity > 0 &&
        (i.product.sub_category === 'motorlar' ||
          i.product.slug.includes('filtre'))
    );
  }, [tankItems]);

  return (
    <group>
      {/* Taban Kumu */}
      <SilisKumMesh
        tankWidth={tankWidth}
        tankDepth={tankDepth}
        tankHeight={tankHeight}
        totalSandKg={totalSandKg}
        sandType={sandType}
      />

      {/* Salyangozlar (Elma & Zebra Nerite â€” Cam iÃ§ yÃ¼zeyi + taban) */}
      <SnailGroup
        specs={snailSpecs}
        innerHalfW={innerHalfW}
        innerHalfD={innerHalfD}
        substrateTopY={substrateTopY}
        waterTopY={waterTopY}
        isCritical={isCritical}
      />

      {/* Bitkiler */}
      {plantPositions.map((p, idx) => (
        <group
          key={`plant-${idx}`}
          position={[p.x, substrateTopY, p.z]}
          scale={p.scale}
        >
          {[0, 1.1, 2.2, 3.3, 4.4].map((angle, leafIdx) => (
            <mesh
              key={leafIdx}
              position={[Math.cos(angle) * 0.015, 0.03 + leafIdx * 0.006, Math.sin(angle) * 0.015]}
              rotation={[0.45, angle, 0.2]}
              castShadow
            >
              <sphereGeometry args={[0.022, 10, 8]} />
              <meshStandardMaterial color={p.color} roughness={0.65} />
            </mesh>
          ))}
        </group>
      ))}

      {/* Filtre Motoru (SaÄŸ arka kÃ¶ÅŸe iÃ§ yÃ¼zey) */}
      {hasFilterMotor && (
        <mesh
          position={[
            innerHalfW - 0.032,
            Math.min(waterTopY - 0.06, substrateTopY + 0.14),
            -innerHalfD + 0.032,
          ]}
          castShadow
        >
          <boxGeometry args={[0.045, 0.12, 0.045]} />
          <meshStandardMaterial color="#1f2937" roughness={0.4} metalness={0.2} />
        </mesh>
      )}
    </group>
  );
}

export function calculateTankZones(
  tankVolume: number,
  totalSandKg: number,
  tankBottomY: number,
  waterSurfaceY: number
): TankZones {
  const maxSandKg = Math.floor((tankVolume * 0.25) / 0.65);
  const clampedSandKg = Math.max(0, totalSandKg);
  const sandVolumeLt = clampedSandKg * 0.65;
  const sandHeightRatio = tankVolume > 0 ? Math.min(sandVolumeLt / tankVolume, 0.30) : 0;
  const totalWaterHeight = waterSurfaceY - tankBottomY;
  const sandHeight = totalWaterHeight * sandHeightRatio;
  const sandTopY = tankBottomY + sandHeight;
  const usableWaterHeight = Math.max(0.1, waterSurfaceY - sandTopY);

  return {
    maxSandKg,
    totalSandKg: clampedSandKg,
    sandVolumeLt,
    sandHeightRatio,
    sandHeight,
    sandTopY,
    usableWaterHeight,
    zones: {
      dip: [sandTopY + 0.08, sandTopY + usableWaterHeight * 0.33],
      orta: [sandTopY + usableWaterHeight * 0.33, sandTopY + usableWaterHeight * 0.66],
      ust: [sandTopY + usableWaterHeight * 0.66, waterSurfaceY - 0.08],
    },
  };
}

/**
 * Ã‡ok Parametreli Ortalama Su Rengi ve Optik GÃ¶rÃ¼nÃ¼m HesaplayÄ±cÄ±sÄ±
 * - Temiz Kristal Mavi (#0284c7)
 * - Alg PatlamasÄ± / YeÅŸil Su (#15803d / #4d7c0f)

function computeBlendedWaterVisuals(params: {
  ammonia: number;
  nitrite: number;
  nitrate: number;
  oxygen: number;
  algaeLevel: number;
  tanninScore: number;
  warnings: string[];
}): WaterVisualState {
  const { ammonia, nitrite, oxygen, algaeLevel, tanninScore } = params;

  // Toksisite skoru (0..1)
  const toxicityFactor = Math.min(
    1,
    Math.max(
      ammonia / 1.2,
      nitrite / 2.0,
      oxygen < 4.2 ? (4.2 - oxygen) / 2.5 : 0
    )
  );

  // Alg yeÅŸillenme skoru (0..1)
  const algaeFactor = Math.min(1, algaeLevel / 100);

  // Tanen amber skoru (0..1)
  const tanninFactor = Math.min(1, tanninScore / 10);

  // Temel renk vektÃ¶rleri
  const cleanColor = new THREE.Color('#0284c7');   // Kristal berrak mavi
  const algaeColor = new THREE.Color('#16a34a');   // ZÃ¼mrÃ¼t/yosun yeÅŸili
  const toxicColor = new THREE.Color('#dc2626');   // Toksik kÄ±zÄ±l-turuncu
  const tanninColor = new THREE.Color('#b45309');  // Amber / Blackwater

  // AÄŸÄ±rlÄ±klÄ± renk harmanlamasÄ±
  const finalColor = cleanColor.clone();
  if (tanninFactor > 0.05) {
    finalColor.lerp(tanninColor, tanninFactor * 0.65);
  }
  if (algaeFactor > 0.05) {
    finalColor.lerp(algaeColor, algaeFactor * 0.85);
  }
  if (toxicityFactor > 0.05) {
    finalColor.lerp(toxicColor, toxicityFactor * 0.85);
  }

  const hexColor = `#${finalColor.getHexString()}`;
  const isLethal = ammonia >= 1.0 || nitrite >= 2.0 || oxygen <= 3.2;

  // BulanÄ±klÄ±k (opacity ve roughness)
  const turbidity = Math.max(toxicityFactor, algaeFactor * 0.85, tanninFactor * 0.4);
  const opacity = Number(Math.min(0.42, 0.13 + turbidity * 0.24).toFixed(3));
  const roughness = Number(Math.min(0.45, 0.10 + turbidity * 0.28).toFixed(2));

  if (isLethal) {
    return {
      phase: 'critical',
      color: hexColor,
      opacity,
      roughness,
      label: 'Kritik Toksisite / Hipoksi',
      isLethal: true,
    };
  }
  if (algaeFactor >= 0.45) {
    return {
      phase: 'algae',
      color: hexColor,
      opacity,
      roughness,
      label: `Alg PatlamasÄ± (%${Math.round(algaeLevel)} YeÅŸil Su)`,
      isLethal: false,
    };
  }
  if (toxicityFactor >= 0.35) {
    return {
      phase: 'moderate',
      color: hexColor,
      opacity,
      roughness,
      label: 'Azot / Oksijen DengesizliÄŸi',
      isLethal: false,
    };
  }
  if (tanninFactor >= 0.35) {
    return {
      phase: 'tannin',
      color: hexColor,
      opacity,
      roughness,
      label: 'DoÄŸal Tanenli (Amber Su)',
      isLethal: false,
    };
  }

  return {
    phase: 'clean',
    color: hexColor,
    opacity,
    roughness,
    label: 'Berrak & Dengeli',
    isLethal: false,
  };
}

      } else if (rule.status === 'uyumlu') {
        if (!positiveNotes.includes(rule.reason)) positiveNotes.push(rule.reason);
      }
    }
  });

  const waterVisuals = computeBlendedWaterVisuals({
    ammonia: finalAmmonia,
    nitrite: finalNitrite,
    nitrate: finalNitrate,
    oxygen: finalO2,
    algaeLevel,
    tanninScore,
    warnings,
  });

  return {
    ph: finalPh,
    temperature: 25.0,
    bioLoad: Number(totalBioLoad.toFixed(1)),
    ammonia: finalAmmonia,
    nitrite: finalNitrite,
    nitrate: finalNitrate,
    oxygen: finalO2,
    co2: finalCo2,
    ppm: finalPpm,
    gh: finalGh,
    kh: finalKh,
    totalLumens: Math.round(totalLumens),
    lightHours,
    lumenPerLiter: Number(lumenPerLiter.toFixed(1)),
    algaeLevel,
    waterVisuals,
    warnings,
    positiveNotes,
    isHealthy: warnings.length === 0,
    maxPlants,
    currentPlants,
  };
}

function normalizeCategory(rawCategory: string): Product['category'] {
  const cat = (rawCategory || '').toLowerCase().trim();
  if (cat === 'balik' || cat === 'baliklar' || cat === 'canli' || cat === 'canlilar') {
    return 'canlilar';
  }
  if (cat === 'bitki' || cat === 'bitkiler') return 'bitkiler';
  if (cat === 'kum' || cat === 'kumlar' || cat === 'taban' || cat === 'dekor') return 'kumlar';
  return 'malzemeler';
}

function inferSubCategory(
  slug: string,
  name: string,
  category: Product['category'],
  rawSub?: string
): string {
  if (rawSub && rawSub.trim().length > 0) return rawSub.toLowerCase().trim();
  const s = `${slug} ${name}`.toLowerCase();

  if (category === 'canlilar') {
    if (s.includes('karides') || s.includes('shrimp') || s.includes('neocaridina')) return 'karidesler';
    if (s.includes('salyangoz') || s.includes('snail') || s.includes('midye') || s.includes('nerite') || s.includes('elma')) return 'kabuklular';
    if (s.includes('bocek') || s.includes('bÃ¶cek') || s.includes('su-piresi') || s.includes('artemia')) return 'bocekler';
    return 'baliklar';
  }

  if (category === 'bitkiler') {
    if (s.includes('moss') || s.includes('yosun')) return 'yosunlar';
    if (s.includes('anubias') || s.includes('bucephalandra') || s.includes('fern')) return 'rizom';
    if (s.includes('pistia') || s.includes('salvinia') || s.includes('mercimek') || s.includes('ust')) return 'su-ustu';
    return 'govdeli';
  }

  if (category === 'kumlar') {
    if (s.includes('toprak') || s.includes('soil') || s.includes('aktif')) return 'aktif-toprak';
    if (s.includes('kok') || s.includes('kÃ¶k') || s.includes('mangrow') || s.includes('kaya') || s.includes('tas')) return 'dekor';
    return 'kum';
  }

  if (s.startsWith('led-') || s.includes('isik') || s.includes('Ä±ÅŸÄ±k') || s.includes('aydinlatma')) return 'isiklar';
  if (s.includes('filtre') || s.includes('motor') || s.includes('kafa') || s.includes('hava')) return 'motorlar';
  if (s.includes('isitici') || s.includes('Ä±sÄ±tÄ±cÄ±') || s.includes('co2') || s.includes('termostat')) return 'isitici-co2';
  return 'diger-ekipman';
}

        lumen: Number(item.lumen ?? 0),

          ? 'uc-renk'
          : slug.includes('beyaz-mavi')
          ? 'beyaz-mavi'
          : slug.startsWith('led-')
          ? 'beyaz'
          : undefined,
      };

    const normalizedRules: CompatibilityRule[] = (rulesRes.data || []).map((r: any) => ({
      id: String(r.id),
      item_a_slug: String(r.item_a_slug),
      item_b_slug: String(r.item_b_slug),

}

    });
  }

  const finDiffuse = new THREE.CanvasTexture(diffCanvas);
  finDiffuse.colorSpace = THREE.SRGBColorSpace;
  finDiffuse.needsUpdate = true;

  const finAlpha = new THREE.CanvasTexture(alphaCanvas);
  finAlpha.needsUpdate = true;

  const finBump = new THREE.CanvasTexture(bumpCanvas);
  finBump.needsUpdate = true;

  return { finDiffuse, finAlpha, finBump };
}

  const radialSegments = 26;
  const positions: number[] = [];
  const uvs: number[] = [];
  const indices: number[] = [];

  const zMin = sections[sections.length - 1].z;
  const zMax = sections[0].z;

  for (let s = 0; s < sections.length; s++) {
    const sec = sections[s];
    const u = (zMax - sec.z) / (zMax - zMin);

    for (let r = 0; r <= radialSegments; r++) {
      const theta = (r / radialSegments) * Math.PI * 2;
      const sinT = Math.sin(theta);
      const cosT = Math.cos(theta);

      const x = sinT * sec.rx;

      const v = 0.5 - cosT * 0.5;

      positions.push(x, y, sec.z);
      uvs.push(u, v);

      const a = s * ringStride + r;
      const b = (s + 1) * ringStride + r;

      const d = s * ringStride + (r + 1);

      indices.push(a, b, d);
      indices.push(b, c, d);
    }
  }

  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geo.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  geo.setIndex(indices);
  geo.computeVertexNormals();
  return geo;
}

function buildBettaEyesGeometry(sex: BettaSex): THREE.BufferGeometry {
  const eyeRadius = sex === 'erkek' ? 0.018 : 0.019;
  const eyeX = sex === 'erkek' ? 0.030 : 0.032;
  const eyeY = 0.015;
  const eyeZ = sex === 'erkek' ? 0.168 : 0.156;

  const leftEye = new THREE.SphereGeometry(eyeRadius, 14, 12);
  leftEye.rotateY(Math.PI / 2);
  leftEye.translate(eyeX, eyeY, eyeZ);

  const rightEye = new THREE.SphereGeometry(eyeRadius, 14, 12);
  rightEye.rotateY(-Math.PI / 2);
  rightEye.translate(-eyeX, eyeY, eyeZ);

  const nonIdxLeft = leftEye.toNonIndexed();
  const nonIdxRight = rightEye.toNonIndexed();

  const totalVerts = nonIdxLeft.attributes.position.count + nonIdxRight.attributes.position.count;
  const positions = new Float32Array(totalVerts * 3);
  const normals = new Float32Array(totalVerts * 3);
  const uvs = new Float32Array(totalVerts * 2);

  positions.set(nonIdxLeft.attributes.position.array as Float32Array, 0);
  positions.set(nonIdxRight.attributes.position.array as Float32Array, nonIdxLeft.attributes.position.count * 3);

  normals.set(nonIdxLeft.attributes.normal.array as Float32Array, 0);
  normals.set(nonIdxRight.attributes.normal.array as Float32Array, nonIdxLeft.attributes.normal.count * 3);

  uvs.set(nonIdxLeft.attributes.uv.array as Float32Array, 0);
  uvs.set(nonIdxRight.attributes.uv.array as Float32Array, nonIdxLeft.attributes.uv.count * 2);

  const merged = new THREE.BufferGeometry();
  merged.setAttribute('position', new THREE.BufferAttribute(positions, 3));
  merged.setAttribute('normal', new THREE.BufferAttribute(normals, 3));
  merged.setAttribute('uv', new THREE.BufferAttribute(uvs, 2));
  return merged;
}

function applyBettaSwimShader(
  material: THREE.MeshPhysicalMaterial,
  timeUniform: { value: number },
  finMultiplier: number
) {
  material.onBeforeCompile = (shader) => {
    shader.uniforms.uTime = timeUniform;
    shader.uniforms.uFinAmp = { value: finMultiplier };

    shader.vertexShader =
      `
      uniform float uTime;
      uniform float uFinAmp;
      ` + shader.vertexShader;

    shader.vertexShader = shader.vertexShader.replace(
      '#include <begin_vertex>',
      `
      #include <begin_vertex>
      float normalizedBack = clamp((0.18 - transformed.z) / 0.68, 0.0, 1.4);
      float spineWave = sin(uTime * 5.2 + transformed.z * 11.5) * pow(normalizedBack, 1.65) * 0.048 * uFinAmp;
      transformed.x += spineWave;

      float radialDist = length(vec2(transformed.y, min(0.0, transformed.z + 0.05)));
      float veilRipple = sin(uTime * 4.4 + transformed.y * 18.0 + transformed.z * 14.0) * smoothstep(0.06, 0.28, radialDist) * 0.022 * (uFinAmp - 0.8);
      transformed.x += max(0.0, uFinAmp - 1.0) * veilRipple;

      float pectoralMask = smoothstep(0.046, 0.13, abs(transformed.x));
      transformed.y += sin(uTime * 11.5 + transformed.x * 18.0) * pectoralMask * 0.024;
      transformed.z += cos(uTime * 11.5) * pectoralMask * 0.012;
      `
    );
  };
}

        const c = baseVertexIndex + (ui + 1) * vSteps + (vi + 1);

function buildEyesGeometry(sex: GuppySex): THREE.BufferGeometry {
  const eyeRadius = sex === 'erkek' ? 0.016 : 0.018;

  const leftEye = new THREE.SphereGeometry(eyeRadius, 14, 12);
  leftEye.rotateY(Math.PI / 2);
  leftEye.translate(eyeX, eyeY, eyeZ);

  const rightEye = new THREE.SphereGeometry(eyeRadius, 14, 12);
  rightEye.rotateY(-Math.PI / 2);
  rightEye.translate(-eyeX, eyeY, eyeZ);

  const nonIdxLeft = leftEye.toNonIndexed();
  const nonIdxRight = rightEye.toNonIndexed();

  const totalVerts = nonIdxLeft.attributes.position.count + nonIdxRight.attributes.position.count;
  const positions = new Float32Array(totalVerts * 3);
  const normals = new Float32Array(totalVerts * 3);
  const uvs = new Float32Array(totalVerts * 2);

  positions.set(nonIdxLeft.attributes.position.array as Float32Array, 0);
  positions.set(nonIdxRight.attributes.position.array as Float32Array, nonIdxLeft.attributes.position.count * 3);

  normals.set(nonIdxLeft.attributes.normal.array as Float32Array, 0);
  normals.set(nonIdxRight.attributes.normal.array as Float32Array, nonIdxLeft.attributes.normal.count * 3);

  uvs.set(nonIdxLeft.attributes.uv.array as Float32Array, 0);
  uvs.set(nonIdxRight.attributes.uv.array as Float32Array, nonIdxLeft.attributes.uv.count * 2);

  const merged = new THREE.BufferGeometry();
  merged.setAttribute('position', new THREE.BufferAttribute(positions, 3));
  merged.setAttribute('normal', new THREE.BufferAttribute(normals, 3));
  merged.setAttribute('uv', new THREE.BufferAttribute(uvs, 2));
  return merged;
}

function applySwimVertexShader(
  material: THREE.MeshPhysicalMaterial,
  timeUniform: { value: number },
  finMultiplier: number
) {
  material.onBeforeCompile = (shader) => {
    shader.uniforms.uTime = timeUniform;
    shader.uniforms.uFinAmp = { value: finMultiplier };

    shader.vertexShader =
      `
      uniform float uTime;
      uniform float uFinAmp;
      ` + shader.vertexShader;

    shader.vertexShader = shader.vertexShader.replace(
      '#include <begin_vertex>',
      `
      #include <begin_vertex>
      float normalizedBack = clamp((0.20 - transformed.z) / 0.65, 0.0, 1.4);
      float spineWave = sin(uTime * 8.5 + transformed.z * 14.0) * pow(normalizedBack, 1.7) * 0.042 * uFinAmp;
      transformed.x += spineWave;

      float pectoralMask = smoothstep(0.042, 0.11, abs(transformed.x));
      transformed.y += sin(uTime * 11.0 + transformed.x * 20.0) * pectoralMask * 0.018;
      `
    );
  };
}

function createPlecoSkinTexture(variant: PlecoVariant): {

  const width = 512;

  const dCtx = dCanvas.getContext('2d')!;

  const bCanvas = document.createElement('canvas');
  bCanvas.width = width;

  // AÃ§Ä±k taban (instanceColor ile koyu kahve, altÄ±n-albino veya leopar rengini alÄ±r)
  dCtx.fillStyle = '#e2e8f0';

  // ZÄ±rhlÄ± kemik plakalarÄ± (Scutes) ve benek desenleri
  const spotCount = variant === 'leopar' ? 140 : 95;
  for (let i = 0; i < spotCount; i++) {

    const r = variant === 'leopar' ? 6 + (i % 5) : 3.5 + (i % 3);

    dCtx.fillStyle = variant === 'leopar' ? 'rgba(15, 23, 42, 0.62)' : 'rgba(255, 255, 255, 0.55)';
    dCtx.beginPath();
    dCtx.arc(x, y, r, 0, Math.PI * 2);
    dCtx.fill();

    bCtx.fillStyle = '#d4d4d4';
    bCtx.beginPath();
    bCtx.arc(x, y, r * 0.9, 0, Math.PI * 2);
    bCtx.fill();
  }

  const diffuseMap = new THREE.CanvasTexture(dCanvas);
  diffuseMap.colorSpace = THREE.SRGBColorSpace;
  diffuseMap.needsUpdate = true;

  const bumpMap = new THREE.CanvasTexture(bCanvas);
  bumpMap.needsUpdate = true;

  return { diffuseMap, bumpMap };

function buildPlecoBodyGeometry(): THREE.BufferGeometry {
  // KarÄ±n kÄ±smÄ± dÃ¼z (vantuzla cama/kuma yapÄ±ÅŸabilmesi iÃ§in ryBot kÃ¼Ã§Ã¼k), sÄ±rtÄ± kubbeli zÄ±rhlÄ± kesit
  const sections = [
    { z: 0.23, rx: 0.018, ryTop: 0.012, ryBot: 0.002, y: 0.002 }, // GeniÅŸ yuvarlak burun
    { z: 0.18, rx: 0.048, ryTop: 0.028, ryBot: 0.004, y: 0.004 }, // GeniÅŸ kafa & vantuz aÄŸÄ±z
    { z: 0.11, rx: 0.056, ryTop: 0.038, ryBot: 0.005, y: 0.005 }, // Omuz geniÅŸliÄŸi
    { z: 0.03, rx: 0.050, ryTop: 0.040, ryBot: 0.005, y: 0.005 }, // GÃ¶vde ortasÄ±
    { z: -0.06, rx: 0.036, ryTop: 0.030, ryBot: 0.004, y: 0.004 },
    { z: -0.15, rx: 0.022, ryTop: 0.020, ryBot: 0.004, y: 0.004 },
    { z: -0.23, rx: 0.010, ryTop: 0.016, ryBot: 0.006, y: 0.006 }, // Kuyruk sapÄ±
  ];

  const radialSegments = 22;
  const positions: number[] = [];
  const uvs: number[] = [];
  const indices: number[] = [];

  for (let s = 0; s < sections.length; s++) {
    const sec = sections[s];
    const u = s / (sections.length - 1);
    for (let r = 0; r <= radialSegments; r++) {
      const theta = (r / radialSegments) * Math.PI * 2;
      const sinT = Math.sin(theta);
      const cosT = Math.cos(theta);
      const x = sinT * sec.rx;
      const ry = cosT >= 0 ? sec.ryTop : sec.ryBot;
      const y = sec.y + cosT * ry;
      positions.push(x, y, sec.z);
      uvs.push(u, r / radialSegments);
    }
  }

  const stride = radialSegments + 1;
  for (let s = 0; s < sections.length - 1; s++) {
    for (let r = 0; r < radialSegments; r++) {
      const a = s * stride + r;
      const b = (s + 1) * stride + r;
      const c = (s + 1) * stride + (r + 1);
      const d = s * stride + (r + 1);
      indices.push(a, b, d);
      indices.push(b, c, d);
    }
  }

  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geo.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  geo.setIndex(indices);
  geo.computeVertexNormals();
  return geo;
}

  const height = 128;
  const canvas = document.createElement('canvas');
  canvas.width = width;
  canvas.height = height;
  const ctx = canvas.getContext('2d')!;

  // NÃ¶tr beyaz-sedef taban (instanceColor ile Ã§arpÄ±lÄ±r)
  const grad = ctx.createLinearGradient(0, 0, width, 0);
  grad.addColorStop(0.0, '#f8fafc');

  // Rilli varyantÄ±nda orta segmentlerde ÅŸeffaf bant efekti

    ctx.fillStyle = 'rgba(203, 213, 225, 0.35)';
    ctx.fillRect(width * 0.38, 0, width * 0.28, height);
  }

  // Amano varyantÄ±nda yan Ã§izgi benekleri
  if (variant === 'amano') {
    ctx.fillStyle = 'rgba(71, 85, 105, 0.55)';

      ctx.beginPath();
      ctx.ellipse(x, height * 0.52, 4, 2, 0, 0, Math.PI * 2);
      ctx.fill();
    }
  }

  // Kabuk segment Ã§izgileri (Carapace & Abdomen eklemleri)
  ctx.strokeStyle = 'rgba(15, 23, 42, 0.28)';
  ctx.lineWidth = 2;
  const segments = [0.34, 0.46, 0.58, 0.70, 0.82];
  segments.forEach((u) => {
    ctx.beginPath();
    ctx.moveTo(width * u, 0);
    ctx.quadraticCurveTo(width * (u + 0.03), height * 0.5, width * u, height);
    ctx.stroke();
  });

  // SÄ±rt yarÄ±ÅŸ Ã§izgisi (Racing stripe)
  ctx.fillStyle = 'rgba(255, 255, 255, 0.42)';
  ctx.fillRect(width * 0.15, height * 0.45, width * 0.75, height * 0.10);

  const tex = new THREE.CanvasTexture(canvas);
  tex.colorSpace = THREE.SRGBColorSpace;
  tex.needsUpdate = true;

}

function buildShrimpBodyGeometry(): THREE.BufferGeometry {
  // Karidesin karakteristik kambur/kavisli gÃ¶vde kesitleri (+Z baÅŸ/rostrum, -Z telson kuyruk)
  const sections = [
    { z: 0.24, rx: 0.002, ry: 0.003, y: 0.025 }, // Rostrum sivri uÃ§
    { z: 0.19, rx: 0.018, ry: 0.020, y: 0.022 }, // Ã–n kafa zÄ±rhÄ±
    { z: 0.12, rx: 0.028, ry: 0.032, y: 0.024 }, // Carapace orta
    { z: 0.04, rx: 0.030, ry: 0.034, y: 0.028 }, // Abdomen 1 (kambur tepe)
    { z: -0.04, rx: 0.026, ry: 0.030, y: 0.024 }, // Abdomen 2
    { z: -0.11, rx: 0.020, ry: 0.023, y: 0.012 }, // Abdomen 3 (aÅŸaÄŸÄ± kÄ±vrÄ±m)
    { z: -0.17, rx: 0.014, ry: 0.016, y: -0.004 }, // Abdomen 4
    { z: -0.22, rx: 0.008, ry: 0.010, y: -0.016 }, // Kuyruk sapÄ±
  ];

  const radialSegments = 16;
  const positions: number[] = [];
  const uvs: number[] = [];
  const indices: number[] = [];

  for (let s = 0; s < sections.length; s++) {
    const sec = sections[s];
    const u = s / (sections.length - 1);
    for (let r = 0; r <= radialSegments; r++) {
      const theta = (r / radialSegments) * Math.PI * 2;
      const x = Math.sin(theta) * sec.rx;
      const y = sec.y + Math.cos(theta) * sec.ry;
      positions.push(x, y, sec.z);
      uvs.push(u, r / radialSegments);
    }
  }

  const stride = radialSegments + 1;
  for (let s = 0; s < sections.length - 1; s++) {
    for (let r = 0; r < radialSegments; r++) {
      const a = s * stride + r;
      const b = (s + 1) * stride + r;
      const c = (s + 1) * stride + (r + 1);
      const d = s * stride + (r + 1);
      indices.push(a, b, d);
      indices.push(b, c, d);
    }
  }

  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geo.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  geo.setIndex(indices);
  geo.computeVertexNormals();
  return geo;
}

function buildShrimpAppendagesGeometry(): THREE.BufferGeometry {

  const uvs: number[] = [];
  const indices: number[] = [];

  const addRibbon = (p0: THREE.Vector3, p1: THREE.Vector3, p2: THREE.Vector3, p3: THREE.Vector3) => {
    const base = positions.length / 3;
    [p0, p1, p2, p3].forEach((p, idx) => {

    indices.push(base + 2, base + 3, base + 1);
  };

  // 1. Uzun Ä°ki Ã‡ift Duyarga (Antennae)
  [-1, 1].forEach((side) => {

      new THREE.Vector3(side * 0.008, 0.024, 0.21),

      new THREE.Vector3(side * 0.055, 0.035, 0.36),
      new THREE.Vector3(side * 0.057, 0.032, 0.36)
    );
  });

  // 2. 5 Ã‡ift YÃ¼rÃ¼yÃ¼ÅŸ BacaÄŸÄ± (Pereiopods)
  const legZs = [0.15, 0.11, 0.07, 0.03, -0.01];
  legZs.forEach((zPos, i) => {

      const spread = 0.032 + (i % 2) * 0.006;
      addRibbon(
        new THREE.Vector3(side * 0.016, 0.004, zPos + 0.004),
        new THREE.Vector3(side * 0.016, 0.004, zPos - 0.004),
        new THREE.Vector3(side * spread, -0.036, zPos + 0.012),
        new THREE.Vector3(side * spread, -0.036, zPos + 0.004)
      );
    });
  });

  // 3. Yelpaze Telson ve Uropod Kuyruk
  addRibbon(
    new THREE.Vector3(-0.008, -0.015, -0.215),
    new THREE.Vector3(0.008, -0.015, -0.215),
    new THREE.Vector3(-0.036, -0.032, -0.285),

  geo.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  geo.setIndex(indices);

  return geo;

function buildShrimpEyesGeometry(): THREE.BufferGeometry {
  const left = new THREE.SphereGeometry(0.008, 10, 8);
  left.translate(0.015, 0.028, 0.195);
  const right = new THREE.SphereGeometry(0.008, 10, 8);
  right.translate(-0.015, 0.028, 0.195);

  const lNon = left.toNonIndexed();
  const rNon = right.toNonIndexed();
  const count = lNon.attributes.position.count + rNon.attributes.position.count;
  const pos = new Float32Array(count * 3);
  const norm = new Float32Array(count * 3);

  pos.set(lNon.attributes.position.array as Float32Array, 0);
  pos.set(rNon.attributes.position.array as Float32Array, lNon.attributes.position.count * 3);
  norm.set(lNon.attributes.normal.array as Float32Array, 0);
  norm.set(rNon.attributes.normal.array as Float32Array, lNon.attributes.normal.count * 3);

  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
  geo.setAttribute('normal', new THREE.BufferAttribute(norm, 3));
  return geo;
}

export function buildSnailGeometries(
  species: SnailSpeciesType = 'elma'
): SnailAnatomyGeometries {
  const isNerite = species === 'nerite';

  // 1. KABUK GEOMETRÄ°SÄ° (Logaritmik Sarmal + Vertex Renk Deseni)
  const turns = isNerite ? 2.1 : 2.8;
  const uSteps = 56;
  const vSteps = 24;

  const positions: number[] = [];
  const colors: number[] = [];
  const uvs: number[] = [];
  const indices: number[] = [];

  const baseColor = new THREE.Color(isNerite ? '#d97706' : '#f59e0b');
  const stripeColor = new THREE.Color(isNerite ? '#18181b' : '#b45309');
  const tmpColor = new THREE.Color();

  for (let i = 0; i <= uSteps; i++) {
    const u = i / uSteps;
    const theta = u * turns * Math.PI * 2;
    const growth = Math.pow(u, isNerite ? 0.72 : 0.85);
    const tubeRadius = (isNerite ? 0.011 : 0.014) * growth + 0.0015;
    const spiralRadius = (isNerite ? 0.008 : 0.012) * growth;
    const spireOffset = isNerite
      ? (1 - u) * 0.004
      : (1 - u) * 0.011;

    for (let j = 0; j <= vSteps; j++) {
      const v = j / vSteps;
      const phi = v * Math.PI * 2;

      const localR = spiralRadius + tubeRadius * Math.cos(phi);
      const x = localR * Math.cos(theta);
      const z = localR * Math.sin(theta);
      const y = tubeRadius * Math.sin(phi) + spireOffset + 0.008;

      positions.push(x, y, z);
      uvs.push(u, v);

      if (isNerite) {
        // Zebra Nerite karakteristik siyah-kehribar zikzak Ã§izgiler
        const stripeWave = Math.sin(theta * 5.0 + Math.sin(phi * 2.0) * 0.9);
        const factor = stripeWave > 0.05 ? 1.0 : 0.0;
        tmpColor.copy(baseColor).lerp(stripeColor, factor);
      } else {
        // Elma salyangozu yumuÅŸak bÃ¼yÃ¼me halkalarÄ±
        const band = (Math.sin(theta * 3.0) + 1) * 0.12;
        tmpColor.copy(baseColor).lerp(stripeColor, band);
      }

      colors.push(tmpColor.r, tmpColor.g, tmpColor.b);
    }
  }

  for (let i = 0; i < uSteps; i++) {
    for (let j = 0; j < vSteps; j++) {
      const a = i * (vSteps + 1) + j;
      const b = (i + 1) * (vSteps + 1) + j;
      const c = (i + 1) * (vSteps + 1) + (j + 1);
      const d = i * (vSteps + 1) + (j + 1);

      indices.push(a, b, d);
      indices.push(b, c, d);
    }
  }

  const shellGeometry = new THREE.BufferGeometry();
  shellGeometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  shellGeometry.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));
  shellGeometry.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  shellGeometry.setIndex(indices);
  shellGeometry.computeVertexNormals();

  // 2. AYAK / GÃ–VDE + ANTENLER
  const bodyGeo = new THREE.CapsuleGeometry(
    isNerite ? 0.006 : 0.0075,
    isNerite ? 0.014 : 0.022,

    14
  );
  bodyGeo.rotateX(Math.PI / 2);

  };

import React, { useRef, useMemo } from 'react';
import { useFrame } from '@react-three/fiber';
import * as THREE from 'three';

interface IcFiltreMeshProps {
  tankWidth: number;
  tankHeight: number;
  tankDepth: number;
  sandTopY: number;
  waterSurfaceY: number;
}

export default function IcFiltreMesh({
  tankWidth,
  tankHeight,
  tankDepth,
  sandTopY,
  waterSurfaceY,
}: IcFiltreMeshProps) {
  const bubblesRef = useRef<THREE.InstancedMesh>(null);
  const count = 20;
  const dummy = useMemo(() => new THREE.Object3D(), []);

  // Ä°Ã§ filtrenin tankÄ±n saÄŸ-arka iÃ§ kÃ¶ÅŸesindeki pozisyonu
  const filterX = tankWidth / 2 - 0.35;
  const filterZ = -tankDepth / 2 + 0.35;

  // Filtre gÃ¶vde yÃ¼ksekliÄŸi
  const filterHeight = 1.4;
  // Filtre tabanÄ±nÄ±n tam sandTopY seviyesine oturmasÄ± iÃ§in merkez Y
  const filterCenterY = sandTopY + filterHeight / 2;
  // Filtre Ã§Ä±kÄ±ÅŸ aÄŸzÄ± Y yÃ¼ksekliÄŸi
  const outletY = sandTopY + filterHeight - 0.15;
  // Su yÃ¼zeyi sÄ±nÄ±rÄ±
  const maxWaterY = waterSurfaceY - 0.05;

  const bubbleGeo = useMemo(() => new THREE.SphereGeometry(1, 8, 8), []);
  const bubbleMat = useMemo(
    () => new THREE.MeshBasicMaterial({ color: '#e0f2fe', transparent: true, opacity: 0.75 }),
    []
  );

  useFrame(({ clock }) => {
    if (!bubblesRef.current) return;
    const t = clock.getElapsedTime();
    const maxRise = Math.max(0, maxWaterY - outletY);

    for (let i = 0; i < count; i++) {
      if (maxRise <= 0.02) {
        dummy.scale.set(0, 0, 0);
        dummy.updateMatrix();
        bubblesRef.current.setMatrixAt(i, dummy.matrix);
        continue;
      }

      // KabarcÄ±k filtre aÄŸzÄ±ndan baÅŸlar, tam su yÃ¼zeyi altÄ±nda kaybolur
      const progress = (t * 0.8 + i * (1 / count)) % 1;
      const y = outletY + progress * maxRise;
      const x = filterX + Math.sin(t * 3 + i) * 0.035;
      const z = filterZ + Math.cos(t * 2.5 + i) * 0.035;

      dummy.position.set(x, y, z);
      const scale = 0.025 + (i % 3) * 0.01;
      dummy.scale.set(scale, scale, scale);
      dummy.updateMatrix();
      bubblesRef.current.setMatrixAt(i, dummy.matrix);
    }
    bubblesRef.current.instanceMatrix.needsUpdate = true;
  });

  return (
    <group>
      {/* Filtre GÃ¶vdesi - TabanÄ± tam sandTopY Ã¼zerine oturur */}
      <mesh position={[filterX, filterCenterY, filterZ]} castShadow>
        <cylinderGeometry args={[0.2, 0.2, filterHeight, 16]} />
        <meshStandardMaterial color="#0284c7" roughness={0.35} metalness={0.6} />
      </mesh>

      {/* Filtre Ã‡Ä±kÄ±ÅŸ Nozulu */}
      <mesh position={[filterX, outletY, filterZ]} castShadow>
        <cylinderGeometry args={[0.07, 0.1, 0.18, 12]} />
        <meshStandardMaterial color="#38bdf8" roughness={0.3} />
      </mesh>

      {/* Hava KabarcÄ±ÄŸÄ± ParÃ§acÄ±klarÄ± (Ã‡Ä±kÄ±ÅŸ aÄŸzÄ±ndan su yÃ¼zeyine) */}
      <instancedMesh ref={bubblesRef} args={[bubbleGeo, bubbleMat, count]} />
    </group>
  );
}

import { create } from 'zustand';
import {
  Product,
  TankItem,
  CompatibilityRule,
  fetchCatalogWithStatus,
} from '../lib/supabase/catalogService';
import {
  EcosystemStatus,
  evaluateEcosystem,
  getMaxPlantsForTank,
} from '../lib/simulation/ecosystemEngine';

interface AquariumState {
  tankCapacity: number; // 50, 100 veya 200 Litre
  lightHours: number;   // 0 - 16 saat gÃ¼nlÃ¼k aydÄ±nlatma sÃ¼resi
  catalog: Product[];
  compatibilityRules: CompatibilityRule[];
  catalogSource: 'supabase' | 'error' | 'loading';
  tankItems: TankItem[];
  searchQuery: string;
  selectedCategory: string;
  ecosystemStatus: EcosystemStatus;

  setTankCapacity: (capacity: number) => void;
  setLightHours: (hours: number) => void;
  loadCatalog: () => Promise<void>;
  addToTank: (product: Product) => void;
  decrementFromTank: (productId: string) => void;
  removeFromTank: (productId: string) => void;
  updateQuantity: (productId: string, quantity: number) => void;
  setSearchQuery: (query: string) => void;
  setSelectedCategory: (category: string) => void;
}

export const useAquariumStore = create<AquariumState>((set, get) => ({
  tankCapacity: 50,
  lightHours: 8,
  catalog: [],
  compatibilityRules: [],
  catalogSource: 'loading',
  tankItems: [],
  searchQuery: '',
  selectedCategory: 'canlilar',
  ecosystemStatus: evaluateEcosystem(50, [], [], 8),

  setTankCapacity: (capacity) => {
    const state = get();
    set({
      tankCapacity: capacity,
      ecosystemStatus: evaluateEcosystem(
        capacity,
        state.tankItems,
        state.compatibilityRules,
        state.lightHours
      ),
    });
  },

  setLightHours: (hours) => {
    const clamped = Math.min(16, Math.max(0, hours));
    const state = get();
    set({
      lightHours: clamped,
      ecosystemStatus: evaluateEcosystem(
        state.tankCapacity,
        state.tankItems,
        state.compatibilityRules,
        clamped
      ),
    });
  },

  loadCatalog: async () => {
    const res = await fetchCatalogWithStatus();
    if (res.source === 'error' && res.errorMessage) {
      console.error('[Supabase HatasÄ±]:', res.errorMessage);
    }

    const currentTankItems = get().tankItems;
    const syncedTankItems = currentTankItems.map((item) => {
      const updatedProduct = res.products.find(
        (p) => p.id === item.product.id || p.slug === item.product.slug
      );
      return updatedProduct ? { ...item, product: updatedProduct } : item;
    });

    const state = get();
    set({
      catalog: res.products,
      compatibilityRules: res.rules,
      catalogSource: res.source,
      tankItems: syncedTankItems,
      ecosystemStatus: evaluateEcosystem(
        state.tankCapacity,
        syncedTankItems,
        res.rules,
        state.lightHours
      ),
    });
  },

  addToTank: (product) => {
    const state = get();
    const currentSandKg = state.tankItems
      .filter((i) => i.product.category === 'kumlar')
      .reduce((acc, i) => acc + i.quantity, 0);

    if (product.category === 'bitkiler') {
      if (currentSandKg <= 0) {
        const msg = 'Bitki eklemek iÃ§in Ã¶nce akvaryuma taban kumu eklemelisiniz!';
        const warnings = state.ecosystemStatus.warnings.includes(msg)
          ? state.ecosystemStatus.warnings
          : [...state.ecosystemStatus.warnings, msg];
        set({ ecosystemStatus: { ...state.ecosystemStatus, warnings, isHealthy: false } });
        return;
      }

      const currentPlants = state.tankItems
        .filter((i) => i.product.category === 'bitkiler')
        .reduce((acc, i) => acc + i.quantity, 0);
      const maxPlants = getMaxPlantsForTank(state.tankCapacity);

      if (currentPlants + 1 > maxPlants) {
        const msg = `Maksimum bitki sÄ±nÄ±rÄ±na ulaÅŸÄ±ldÄ±! (${state.tankCapacity}L tabanÄ± iÃ§in max ${maxPlants} adet)`;
        const warnings = state.ecosystemStatus.warnings.includes(msg)
          ? state.ecosystemStatus.warnings
          : [...state.ecosystemStatus.warnings, msg];
        set({ ecosystemStatus: { ...state.ecosystemStatus, warnings, isHealthy: false } });
        return;
      }
    }

    if (product.category === 'kumlar') {
      const maxSandKg = Math.floor((state.tankCapacity * 0.25) / 0.65);
      if (currentSandKg + 1 > maxSandKg) {
        const msg = `Maksimum kum sÄ±nÄ±rÄ±na ulaÅŸÄ±ldÄ±! (${state.tankCapacity}L iÃ§in max ${maxSandKg} kg)`;
        const warnings = state.ecosystemStatus.warnings.includes(msg)
          ? state.ecosystemStatus.warnings
          : [...state.ecosystemStatus.warnings, msg];
        set({ ecosystemStatus: { ...state.ecosystemStatus, warnings, isHealthy: false } });
        return;
      }
    }

    const existing = state.tankItems.find((item) => item.product.id === product.id);
    const newItems = existing
      ? state.tankItems.map((item) =>
          item.product.id === product.id ? { ...item, quantity: item.quantity + 1 } : item
        )
      : [...state.tankItems, { product, quantity: 1 }];

    set({
      tankItems: newItems,
      ecosystemStatus: evaluateEcosystem(
        state.tankCapacity,
        newItems,
        state.compatibilityRules,
        state.lightHours
      ),
    });
  },

  decrementFromTank: (productId) => {
    const state = get();
    const existing = state.tankItems.find((item) => item.product.id === productId);
    if (!existing) return;

    if (existing.quantity <= 1) {
      state.removeFromTank(productId);
    } else {
      state.updateQuantity(productId, existing.quantity - 1);
    }
  },

  removeFromTank: (productId) => {
    const state = get();
    const newItems = state.tankItems.filter((item) => item.product.id !== productId);
    set({
      tankItems: newItems,
      ecosystemStatus: evaluateEcosystem(
        state.tankCapacity,
        newItems,
        state.compatibilityRules,
        state.lightHours
      ),
    });
  },

  updateQuantity: (productId, quantity) => {
    const state = get();
    if (quantity <= 0) {
      state.removeFromTank(productId);
      return;
    }

    const targetItem = state.tankItems.find((item) => item.product.id === productId);
    if (!targetItem) return;

    if (targetItem.product.category === 'bitkiler') {
      const otherPlants = state.tankItems
        .filter((i) => i.product.category === 'bitkiler' && i.product.id !== productId)
        .reduce((sum, i) => sum + i.quantity, 0);
      const maxPlants = getMaxPlantsForTank(state.tankCapacity);
      if (otherPlants + quantity > maxPlants) return;
    }

    if (targetItem.product.category === 'kumlar') {
      const otherSandKg = state.tankItems
        .filter((i) => i.product.category === 'kumlar' && i.product.id !== productId)
        .reduce((sum, i) => sum + i.quantity, 0);
      const maxSandKg = Math.floor((state.tankCapacity * 0.25) / 0.65);
      if (otherSandKg + quantity > maxSandKg) return;
    }

    const newItems = state.tankItems.map((item) =>
      item.product.id === productId ? { ...item, quantity } : item
    );

    set({
      tankItems: newItems,
      ecosystemStatus: evaluateEcosystem(
        state.tankCapacity,
        newItems,
        state.compatibilityRules,
        state.lightHours
      ),
    });
  },

  setSearchQuery: (searchQuery) => set({ searchQuery }),
  setSelectedCategory: (selectedCategory) => set({ selectedCategory }),
}));
